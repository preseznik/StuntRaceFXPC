cmake_minimum_required(VERSION 3.24)
project(StuntRaceFXPC VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The initial native tooling targets Windows x64.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use an x64 compiler.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(srf_core STATIC src/rom.cpp src/random.cpp src/mesh.cpp src/scene.cpp src/championship.cpp
    src/world.cpp src/streaming.cpp src/vehicle_math.cpp src/vehicle.cpp src/race.cpp src/native_race.cpp
    src/body_collision.cpp src/audio.cpp src/music_renderer.cpp src/race_audio.cpp src/presentation.cpp
    src/environment.cpp src/scenery.cpp src/backdrop.cpp src/sunset_rocks.cpp src/night_owl_cones.cpp
    src/water_spray.cpp src/white_land_breakables.cpp src/night_cruise_bonus.cpp src/rain.cpp
    src/sky_ramp_aircraft.cpp src/speed_bonus.cpp src/trailer_loading.cpp
    src/rock_field.cpp src/rock_field_rules.cpp src/stunt_scores.cpp)
target_include_directories(srf_core PUBLIC include)
target_link_libraries(srf_core PRIVATE bcrypt)
target_compile_definitions(srf_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
    target_compile_options(srf_core PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(srf_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(srf_romtool src/romtool.cpp)
target_link_libraries(srf_romtool PRIVATE srf_core)
if(MINGW)
    target_link_options(srf_romtool PRIVATE -municode -static)
endif()

        message(FATAL_ERROR "Generate assets with srf_romtool before configuring the preview.")
    endif()
    set(SRF_SDL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.tools/sdl3-3.4.16/SDL3-3.4.16/x86_64-w64-mingw32" CACHE PATH "Pinned SDL3 development package")

    set(SRF_XR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.tools/openxr-1.1.53")
    add_library(srf_openxr STATIC src/openxr_drive.cpp)

    target_include_directories(srf_preview PRIVATE "${SRF_GENERATED_DATA}")
    target_link_libraries(srf_preview PRIVATE srf_core srf_openxr SDL3::SDL3 d3d11 dxgi d3dcompiler windowscodecs ole32)
    add_custom_command(TARGET srf_preview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:srf_preview>)
    if(MINGW)
        target_link_options(srf_preview PRIVATE -municode -static)
    endif()
    install(TARGETS srf_preview RUNTIME DESTINATION .)
    install(FILES $<TARGET_FILE:SDL3::SDL3-shared> DESTINATION .)
    if(BUILD_TESTING)
